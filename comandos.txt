CREATE TABLE teste (
 id INT PRIMARY KEY,
 nome VARCHAR(50),
 valor DECIMAL(10,2)
);

# =========================

# boa prática

SELECT nome FROM teste;

SELECT * FROM teste WHERE valor > 1500;

SELECT * FROM teste ORDER BY valor DESC;

# =========================
# FOREIGN KEY (DEMO)
# =========================
CREATE TABLE depto (
 id INT PRIMARY KEY,
 nome VARCHAR(50)
);

CREATE TABLE emp (
 id INT PRIMARY KEY,
 nome VARCHAR(50),
 depto_id INT
);

# adicionar FK
ALTER TABLE emp
ADD CONSTRAINT fk_emp_depto
FOREIGN KEY (depto_id)
REFERENCES depto(id);

# erro esperado
INSERT INTO emp VALUES (1, 'Erro', 999);

# correto
INSERT INTO depto VALUES (1, 'TI');
INSERT INTO emp VALUES (1, 'Ana', 1);

# =========================
# CASCADE
# =========================
